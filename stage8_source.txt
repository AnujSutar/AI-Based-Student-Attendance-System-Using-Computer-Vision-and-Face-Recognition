def show_capture_face() -> None:
    st.title("Capture Face Samples")
    st.caption("Capture three clear samples for each student. Only one face should be visible.")
    students = get_student_options()
    if not students:
        st.warning("Register a student before capturing face samples.")
        return

    student_lookup = {
        f"{student['name']} — {student['roll_number']} ({student['student_id']})": student
        for student in students
    }
    selected_label = st.selectbox("Select student", list(student_lookup))
    student = student_lookup[selected_label]
    current_count = count_face_samples(student["student_id"])
    st.progress(current_count / MAX_FACE_SAMPLES, text=f"Samples captured: {current_count}/{MAX_FACE_SAMPLES}")
    st.info(
        "Take three photos with slightly different natural angles and expressions. Keep your face clear, "
        "well lit, and alone in the frame."
    )

    captured_image = st.camera_input("Take a face photo", key=f"camera_{student['student_id']}")
    if captured_image is not None:
        image_hash = hash(captured_image.getvalue())
        processed_key = f"processed_{student['student_id']}"
        if st.session_state.get(processed_key) != image_hash:
            st.session_state[processed_key] = image_hash
            success, message, _ = assess_and_save_face(captured_image.getvalue(), student["student_id"])
            if success:
                st.success(message)
                st.rerun()
            else:
                st.error(message)

    st.divider()
    st.subheader("Or upload photos from this computer")
    uploaded_images = st.file_uploader(
        "Choose up to 3 clear face photos",
        type=["jpg", "jpeg", "png"],
        accept_multiple_files=True,
        key=f"upload_{student['student_id']}",
    )
    if uploaded_images:
        st.caption("Only new photos are processed. The app still stores a maximum of 3 samples per student.")
        if len(uploaded_images) > 3:
            st.warning("Choose no more than 3 photos at one time.")
        elif st.button("Validate and save uploaded photos", key=f"save_uploads_{student['student_id']}"):
            processed_key = f"uploaded_hashes_{student['student_id']}"
            processed_hashes = st.session_state.setdefault(processed_key, set())
            saved_count = 0
            for uploaded_image in uploaded_images:
                image_bytes = uploaded_image.getvalue()
                image_hash = hash(image_bytes)
                if image_hash in processed_hashes:
                    continue
                processed_hashes.add(image_hash)
                success, message, _ = assess_and_save_face(image_bytes, student["student_id"])
                if success:
                    saved_count += 1
                    st.success(f"{uploaded_image.name}: {message}")
                else:
                    st.error(f"{uploaded_image.name}: {message}")
            if saved_count:
                new_total = count_face_samples(student["student_id"])
                st.info(
                    f"Upload complete: {saved_count} photo(s) saved. "
                    f"This student now has {new_total}/{MAX_FACE_SAMPLES} face samples."
                )

    if current_count >= MIN_FACE_SAMPLES:
        st.success("This student has enough samples for the recognition-training stage.")

def show_live_recognition() -> None:
    st.title("Live Face Recognition")
    st.caption("Stage 4: identify known faces with the laptop webcam. Attendance is not recorded yet.")
    st.info(
        "This uses local LBPH face recognition. It compares live face patterns with the three "
        "saved samples; no face data is sent online."
    )

    left, right = st.columns([1, 2])
    with left:
        if st.button("Train / retrain recognition model", type="primary"):
            success, message = train_recognizer()
            if success:
                st.success(message)
            else:
                st.error(message)
    with right:
        st.caption("Retrain after adding or deleting face samples.")

    duration = st.slider("Live camera duration (seconds)", min_value=5, max_value=30, value=15)
    if st.button("Start live recognition"):
        recognizer, label_map, error = load_recognizer()
        if error:
            st.error(error)
            return

        camera = cv2.VideoCapture(CAMERA_INDEX, cv2.CAP_DSHOW)
        if not camera.isOpened():
            camera.release()
            st.error("Webcam unavailable. Check camera permissions and close other apps using the camera.")
            return

        camera_view = st.empty()
        status_box = st.empty()
        recognized_names: set[str] = set()
        end_time = time.time() + duration
        try:
            while time.time() < end_time:
                success, frame = camera.read()
                if not success:
                    st.error("A camera frame could not be read. Try reconnecting the webcam.")
                    break
                results = recognize_faces(frame, recognizer, label_map)
                for result in results:
                    x, y, width, height = result["box"]
                    color = (0, 180, 0) if result["result"] == "Recognized" else (0, 165, 255)
                    if result["result"] == "Unknown":
                        color = (0, 0, 255)
                    label = f"{result['name']} | {result['similarity']}%"
                    cv2.rectangle(frame, (x, y), (x + width, y + height), color, 2)
                    cv2.putText(frame, label, (x, max(y - 10, 25)), cv2.FONT_HERSHEY_SIMPLEX, 0.6, color, 2)
                    if result["result"] == "Recognized":
                        recognized_names.add(f"{result['name']} ({result['roll_number']})")
                camera_view.image(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB), channels="RGB", use_container_width=True)
                status_box.caption(f"Scanning... {max(0, int(end_time - time.time()))} second(s) remaining.")
        except RuntimeError as error:
            st.error(str(error))
        finally:
            camera.release()

        if recognized_names:
            st.success("Recognized during this session: " + ", ".join(sorted(recognized_names)))
        else:
            st.warning("No high-confidence registered face was recognized during this session.")
        st.info("Live recognition session complete. Stage 5 will add automatic attendance marking.")


def show_live_attendance() -> None:
    st.title("Live Attendance")
    st.caption("Recognized students are marked Present automatically, once per day.")
    st.warning(
        "Only green, high-confidence recognitions are marked. Orange uncertain and red unknown faces are not recorded."
    )
    duration = st.slider("Attendance camera duration (seconds)", min_value=5, max_value=30, value=15, key="attendance_duration")
    if st.button("Start attendance camera", type="primary"):
        recognizer, label_map, error = load_recognizer()
        if error:
            st.error(error)
            return

        camera = cv2.VideoCapture(CAMERA_INDEX, cv2.CAP_DSHOW)
        if not camera.isOpened():
            camera.release()
            st.error("Webcam unavailable. Check camera permissions and close other apps using the camera.")
            return

        camera_view = st.empty()
        status_box = st.empty()
        result_box = st.empty()
        handled_student_ids: set[str] = set()
        session_messages: list[str] = []
        end_time = time.time() + duration
        try:
            while time.time() < end_time:
                success, frame = camera.read()
                if not success:
                    st.error("A camera frame could not be read. Try reconnecting the webcam.")
                    break
                results = recognize_faces(frame, recognizer, label_map)
                for result in results:
                    x, y, width, height = result["box"]
                    color = (0, 180, 0) if result["result"] == "Recognized" else (0, 165, 255)
                    if result["result"] == "Unknown":
                        color = (0, 0, 255)
                    label = f"{result['name']} | {result['similarity']}%"
                    cv2.rectangle(frame, (x, y), (x + width, y + height), color, 2)
                    cv2.putText(frame, label, (x, max(y - 10, 25)), cv2.FONT_HERSHEY_SIMPLEX, 0.6, color, 2)

                    student_id = result["student_id"]
                    if result["result"] == "Recognized" and student_id and student_id not in handled_student_ids:
                        handled_student_ids.add(student_id)
                        created, message = mark_attendance(student_id, result["similarity"])
                        session_messages.append(message)

                camera_view.image(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB), channels="RGB", use_container_width=True)
                status_box.caption(f"Attendance scanning... {max(0, int(end_time - time.time()))} second(s) remaining.")
                if session_messages:
                    result_box.info("Latest result: " + session_messages[-1])
        except RuntimeError as error:
            st.error(str(error))
        finally:
            camera.release()

        if session_messages:
            st.success("Attendance session complete.")
            for message in session_messages:
                st.write(f"- {message}")
        else:
            st.warning("No high-confidence registered face was detected; no attendance was marked.")

"""Simple local admin authentication for the college prototype."""

from datetime import datetime

from database import get_connection, hash_password, verify_password


def authenticate(username: str, password: str) -> bool:
    """Verify an admin username and password against its salted stored hash."""
    with get_connection() as connection:
        admin = connection.execute(
            "SELECT password_hash FROM admin WHERE username = ?", (username.strip(),)
        ).fetchone()
    return bool(admin and verify_password(password, admin["password_hash"]))


def change_password(username: str, current_password: str, new_password: str) -> tuple[bool, str]:
    """Change the logged-in admin password after current-password verification."""
    if len(new_password) < 6:
        return False, "Use a password with at least 6 characters."
    if not authenticate(username, current_password):
        return False, "Current password is incorrect."
    with get_connection() as connection:
        connection.execute(
            "UPDATE admin SET password_hash = ?, created_at = ? WHERE username = ?",
            (hash_password(new_password), datetime.now().isoformat(timespec="seconds"), username),
        )
    return True, "Password changed successfully."

"""Persistent application settings stored locally in SQLite."""

from config import CAMERA_INDEX, DUPLICATE_INTERVAL_MINUTES, LOW_ATTENDANCE_THRESHOLD, MAX_RECOGNITION_DISTANCE
from database import get_connection


DEFAULT_SETTINGS = {
    "app_name": "AI-Based Student Attendance System",
    "camera_index": str(CAMERA_INDEX),
    "max_recognition_distance": str(MAX_RECOGNITION_DISTANCE),
    "duplicate_interval_minutes": str(DUPLICATE_INTERVAL_MINUTES),
    "low_attendance_threshold": str(LOW_ATTENDANCE_THRESHOLD),
}


def initialize_settings() -> None:
    with get_connection() as connection:
        for key, value in DEFAULT_SETTINGS.items():
            connection.execute(
                "INSERT OR IGNORE INTO application_settings (setting_key, setting_value) VALUES (?, ?)",
                (key, value),
            )


def get_settings() -> dict[str, str]:
    initialize_settings()
    with get_connection() as connection:
        rows = connection.execute("SELECT setting_key, setting_value FROM application_settings").fetchall()
    stored = {row["setting_key"]: row["setting_value"] for row in rows}
    return {**DEFAULT_SETTINGS, **stored}


def update_settings(settings: dict[str, str]) -> tuple[bool, str]:
    """Validate and persist configurable settings."""
    try:
        camera_index = int(settings["camera_index"])
        max_distance = float(settings["max_recognition_distance"])
        duplicate_interval = int(settings["duplicate_interval_minutes"])
        low_threshold = float(settings["low_attendance_threshold"])
    except (KeyError, TypeError, ValueError):
        return False, "Settings must contain valid numeric values."
    if camera_index < 0 or not 0 < max_distance <= 100 or duplicate_interval < 1 or not 0 <= low_threshold <= 100:
        return False, "Enter valid values for all settings."

    values = {
        "app_name": settings["app_name"].strip() or DEFAULT_SETTINGS["app_name"],
        "camera_index": str(camera_index),
        "max_recognition_distance": str(max_distance),
        "duplicate_interval_minutes": str(duplicate_interval),
        "low_attendance_threshold": str(low_threshold),
    }
    with get_connection() as connection:
        for key, value in values.items():
            connection.execute(
                "INSERT INTO application_settings (setting_key, setting_value) VALUES (?, ?) "
                "ON CONFLICT(setting_key) DO UPDATE SET setting_value = excluded.setting_value",
                (key, value),
            )
    return True, "Settings saved successfully."

def create_directories() -> None:
    """Create local folders that the application needs."""
    for directory in (DATABASE_DIR, FACE_DATA_DIR, MODEL_DIR, CAPTURED_FACES_DIR, REPORTS_DIR):
        directory.mkdir(parents=True, exist_ok=True)


def get_connection() -> sqlite3.Connection:
    """Return a SQLite connection with foreign-key protection enabled."""
    create_directories()
    connection = sqlite3.connect(DATABASE_PATH)
    connection.row_factory = sqlite3.Row
    connection.execute("PRAGMA foreign_keys = ON")
    return connection


def hash_password(password: str, salt: bytes | None = None) -> str:
    """Create a salted PBKDF2 password hash for local admin authentication."""
    salt = salt or os.urandom(16)
    password_hash = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, 100_000)
    return f"{salt.hex()}${password_hash.hex()}"


def verify_password(password: str, stored_hash: str) -> bool:
    """Check a password against a PBKDF2 hash without exposing plain text."""
    salt_hex, expected_hash = stored_hash.split("$", maxsplit=1)
    actual_hash = hash_password(password, bytes.fromhex(salt_hex)).split("$", maxsplit=1)[1]
    return hmac.compare_digest(actual_hash, expected_hash)


def initialize_database() -> None:
    """Create all Stage 1 tables and seed one development admin account."""
    create_directories()
    with get_connection() as connection:
        connection.executescript(
            """
            CREATE TABLE IF NOT EXISTS admin (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT NOT NULL UNIQUE,
                password_hash TEXT NOT NULL,
                created_at TEXT NOT NULL
            );

            CREATE TABLE IF NOT EXISTS students (
                student_id TEXT PRIMARY KEY,
                roll_number TEXT NOT NULL UNIQUE,
                name TEXT NOT NULL,
                class_name TEXT NOT NULL,
                division TEXT NOT NULL,
                department TEXT NOT NULL,
                email TEXT,
                created_at TEXT NOT NULL
            );

            CREATE TABLE IF NOT EXISTS face_embeddings (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                student_id TEXT NOT NULL,
                embedding BLOB NOT NULL,
                created_at TEXT NOT NULL,
                FOREIGN KEY (student_id) REFERENCES students(student_id) ON DELETE CASCADE
            );

            CREATE TABLE IF NOT EXISTS face_samples (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                student_id TEXT NOT NULL,
                file_path TEXT NOT NULL UNIQUE,
                captured_at TEXT NOT NULL,
                quality_score REAL NOT NULL,
                FOREIGN KEY (student_id) REFERENCES students(student_id) ON DELETE CASCADE
            );

            CREATE TABLE IF NOT EXISTS attendance (
                attendance_id INTEGER PRIMARY KEY AUTOINCREMENT,
                student_id TEXT NOT NULL,
                attendance_date TEXT NOT NULL,
                attendance_time TEXT NOT NULL,
                status TEXT NOT NULL CHECK(status IN ('Present', 'Absent')),
                confidence REAL,
                created_at TEXT NOT NULL,
                UNIQUE(student_id, attendance_date),
                FOREIGN KEY (student_id) REFERENCES students(student_id) ON DELETE CASCADE
            );

            CREATE TABLE IF NOT EXISTS recognition_logs (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                student_id TEXT,
                recognized_name TEXT,
                log_date TEXT NOT NULL,
                log_time TEXT NOT NULL,
                result TEXT NOT NULL,
                confidence REAL,
                status TEXT NOT NULL,
                FOREIGN KEY (student_id) REFERENCES students(student_id) ON DELETE SET NULL
            );

            CREATE TABLE IF NOT EXISTS application_settings (
                setting_key TEXT PRIMARY KEY,
                setting_value TEXT NOT NULL
            );
            """
        )

        existing_admin = connection.execute(
            "SELECT id FROM admin WHERE username = ?", (DEFAULT_ADMIN_USERNAME,)
        ).fetchone()
        if existing_admin is None:
            connection.execute(
                "INSERT INTO admin (username, password_hash, created_at) VALUES (?, ?, ?)",
                (DEFAULT_ADMIN_USERNAME, hash_password(DEFAULT_ADMIN_PASSWORD), datetime.now().isoformat()),
            )


def get_table_counts() -> dict[str, int]:
    """Return a small database health summary for the Stage 1 screen."""
    with get_connection() as connection:
        return {
            "Students": connection.execute("SELECT COUNT(*) FROM students").fetchone()[0],
            "Face samples": connection.execute("SELECT COUNT(*) FROM face_samples").fetchone()[0],
            "Attendance records": connection.execute("SELECT COUNT(*) FROM attendance").fetchone()[0],
            "Recognition logs": connection.execute("SELECT COUNT(*) FROM recognition_logs").fetchone()[0],
        }

"""Local face detection, quality checks, sample storage, and LBPH recognition."""

import json
from datetime import datetime
from pathlib import Path

import cv2
import numpy as np

from config import (
    FACE_DATA_DIR,
    LABELS_PATH,
    MAX_FACE_SAMPLES,
    MAX_RECOGNITION_DISTANCE,
    MIN_FACE_SAMPLES,
    MODEL_PATH,
)
from database import get_connection


def get_face_detector() -> cv2.CascadeClassifier:
    """Load OpenCV's bundled Haar-cascade face detector."""
    cascade_path = Path(cv2.data.haarcascades) / "haarcascade_frontalface_default.xml"
    detector = cv2.CascadeClassifier(str(cascade_path))
    if detector.empty():
        raise RuntimeError("The OpenCV face detector could not be loaded.")
    return detector


def count_face_samples(student_id: str) -> int:
    with get_connection() as connection:
        return connection.execute(
            "SELECT COUNT(*) FROM face_samples WHERE student_id = ?", (student_id,)
        ).fetchone()[0]


def get_face_sample_summary() -> list[dict]:
    """Return sample counts so the operator can see training readiness."""
    with get_connection() as connection:
        rows = connection.execute(
            """
            SELECT s.student_id, s.roll_number, s.name, COUNT(fs.id) AS sample_count
            FROM students AS s
            LEFT JOIN face_samples AS fs ON fs.student_id = s.student_id
            GROUP BY s.student_id, s.roll_number, s.name
            ORDER BY s.name COLLATE NOCASE
            """
        ).fetchall()
    return [dict(row) for row in rows]


def delete_face_samples(student_id: str) -> tuple[bool, str]:
    """Permanently remove one student's saved face photos and sample records."""
    with get_connection() as connection:
        rows = connection.execute(
            "SELECT file_path FROM face_samples WHERE student_id = ?", (student_id,)
        ).fetchall()

    if not rows:
        return False, "No saved face data was found for this student."

    face_root = FACE_DATA_DIR.resolve()
    paths = [Path(row["file_path"]).resolve() for row in rows]
    invalid_paths = [path for path in paths if not path.is_relative_to(face_root)]
    if invalid_paths:
        return False, "Face data could not be deleted because an unsafe file path was detected."

    failed_paths = []
    for path in paths:
        try:
            if path.exists():
                path.unlink()
        except OSError:
            failed_paths.append(path.name)
    if failed_paths:
        return False, "Some local face photos could not be deleted. Close any app using them and try again."

    with get_connection() as connection:
        connection.execute("DELETE FROM face_samples WHERE student_id = ?", (student_id,))

    # Remove the now-empty per-student folder when it is safe to do so.
    for parent in {path.parent for path in paths}:
        if parent.is_relative_to(face_root):
            try:
                parent.rmdir()
            except OSError:
                pass
    return True, "Face data deleted successfully. This student must be captured again before recognition."


def _create_lbph_recognizer():
    """Create OpenCV's local LBPH recognizer."""
    if not hasattr(cv2, "face") or not hasattr(cv2.face, "LBPHFaceRecognizer_create"):
        raise RuntimeError("LBPH recognition is unavailable. Install opencv-contrib-python and restart the app.")
    return cv2.face.LBPHFaceRecognizer_create()


def train_recognizer() -> tuple[bool, str]:
    """Train and save an offline LBPH model from students with three samples."""
    try:
        recognizer = _create_lbph_recognizer()
    except RuntimeError as error:
        return False, str(error)

    with get_connection() as connection:
        rows = connection.execute(
            """
            SELECT fs.student_id, fs.file_path, s.name, s.roll_number
            FROM face_samples AS fs JOIN students AS s ON s.student_id = fs.student_id
            ORDER BY fs.student_id, fs.id
            """
        ).fetchall()

    by_student: dict[str, list] = {}
    for row in rows:
        by_student.setdefault(row["student_id"], []).append(row)
    eligible = {student_id: samples for student_id, samples in by_student.items() if len(samples) >= MIN_FACE_SAMPLES}
    if not eligible:
        return False, f"Capture {MIN_FACE_SAMPLES} face samples for at least one student before training."

    images: list[np.ndarray] = []
    labels: list[int] = []
    label_map: dict[str, dict[str, str]] = {}
    for numeric_label, student_id in enumerate(sorted(eligible)):
        student_samples = eligible[student_id]
        label_map[str(numeric_label)] = {
            "student_id": student_id,
            "name": student_samples[0]["name"],
            "roll_number": student_samples[0]["roll_number"],
        }
        for sample in student_samples:
            image = cv2.imread(sample["file_path"], cv2.IMREAD_GRAYSCALE)
            if image is not None:
                images.append(cv2.resize(image, (200, 200)))
                labels.append(numeric_label)
    if not images:
        return False, "No readable face samples were found. Capture the face data again."

    recognizer.train(images, np.array(labels, dtype=np.int32))
    recognizer.write(str(MODEL_PATH))
    LABELS_PATH.write_text(json.dumps(label_map, indent=2), encoding="utf-8")
    return True, f"Recognition model trained with {len(images)} samples from {len(label_map)} student(s)."


def load_recognizer():
    """Load the saved local model and its student-label map."""
    if not MODEL_PATH.exists() or not LABELS_PATH.exists():
        return None, None, "Train the recognition model before starting the camera."
    try:
        recognizer = _create_lbph_recognizer()
        recognizer.read(str(MODEL_PATH))
        return recognizer, json.loads(LABELS_PATH.read_text(encoding="utf-8")), None
    except (RuntimeError, cv2.error, json.JSONDecodeError):
        return None, None, "The saved recognition model could not be loaded. Train it again."


def recognize_faces(frame: np.ndarray, recognizer, label_map: dict) -> list[dict]:
    """Detect every face in a frame and return its recognition result."""
    detector = get_face_detector()
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    faces = detector.detectMultiScale(gray, scaleFactor=1.15, minNeighbors=6, minSize=(90, 90))
    results = []
    for x, y, width, height in faces:
        face = cv2.resize(gray[y : y + height, x : x + width], (200, 200))
        predicted_label, distance = recognizer.predict(face)
        similarity = round(max(0.0, 100.0 - float(distance)), 1)
        student = label_map.get(str(predicted_label))
        if student and distance <= MAX_RECOGNITION_DISTANCE:
            result = {**student, "result": "Recognized", "similarity": similarity}
        elif student and distance <= MAX_RECOGNITION_DISTANCE + 15:
            result = {**student, "result": "Recognition uncertain", "similarity": similarity}
        else:
            result = {"student_id": None, "name": "Unknown", "roll_number": "—", "result": "Unknown", "similarity": similarity}
        result["box"] = (int(x), int(y), int(width), int(height))
        results.append(result)
    return results


def assess_and_save_face(image_bytes: bytes, student_id: str) -> tuple[bool, str, float | None]:
    """Validate one camera image, crop its face, and store it locally for training."""
    sample_count = count_face_samples(student_id)
    if sample_count >= MAX_FACE_SAMPLES:
        return False, f"This student already has {MAX_FACE_SAMPLES} face samples.", None

    image_array = np.frombuffer(image_bytes, dtype=np.uint8)
    image = cv2.imdecode(image_array, cv2.IMREAD_COLOR)
    if image is None:
        return False, "The captured image could not be read. Please take another photo.", None

    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    try:
        detector = get_face_detector()
    except RuntimeError:
        return (
            False,
            "Face detection is not available in this OpenCV installation. "
            "Reinstall opencv-contrib-python using the command in the project instructions.",
            None,
        )
    faces = detector.detectMultiScale(gray, scaleFactor=1.15, minNeighbors=6, minSize=(90, 90))

    if len(faces) == 0:
        return False, "No clear face detected. Face the camera and improve the lighting.", None
    if len(faces) > 1:
        return False, "Multiple faces detected. Ensure only the selected student is in the frame.", None

    x, y, width, height = faces[0]
    face = gray[y : y + height, x : x + width]
    brightness = float(face.mean())
    sharpness = float(cv2.Laplacian(face, cv2.CV_64F).var())

    if brightness < 45:
        return False, "The image is too dark. Move to a better-lit area and try again.", None
    if brightness > 220:
        return False, "The image is overexposed. Reduce direct light and try again.", None
    if sharpness < 45:
        return False, "The face image is blurry. Hold still and capture another image.", None

    face = cv2.resize(face, (200, 200))
    quality_score = round(min(100, (sharpness / 4) + min(brightness, 180) / 3), 1)
    student_folder = FACE_DATA_DIR / student_id
    student_folder.mkdir(parents=True, exist_ok=True)
    filename = f"face_{datetime.now().strftime('%Y%m%d_%H%M%S_%f')}.jpg"
    output_path = student_folder / filename

    if not cv2.imwrite(str(output_path), face):
        return False, "The face sample could not be stored. Please try again.", None

    try:
        with get_connection() as connection:
            connection.execute(
                """
                INSERT INTO face_samples (student_id, file_path, captured_at, quality_score)
                VALUES (?, ?, ?, ?)
                """,
                (student_id, str(output_path), datetime.now().isoformat(timespec="seconds"), quality_score),
            )
    except Exception:
        output_path.unlink(missing_ok=True)
        return False, "The sample was not saved to the database. Please try again.", None

    current_total = sample_count + 1
    return True, f"Face sample {current_total}/{MAX_FACE_SAMPLES} saved (quality score: {quality_score}).", quality_score

"""Attendance recording and attendance-record queries."""

import sqlite3
from datetime import datetime

from database import get_connection


def mark_attendance(student_id: str, confidence: float) -> tuple[bool, str]:
    """Record Present once per student per day; return whether a new record was made."""
    now = datetime.now()
    attendance_date = now.date().isoformat()
    attendance_time = now.strftime("%H:%M:%S")
    try:
        with get_connection() as connection:
            student = connection.execute(
                "SELECT name, roll_number FROM students WHERE student_id = ?", (student_id,)
            ).fetchone()
            if student is None:
                return False, "Recognized student profile was not found."

            connection.execute(
                """
                INSERT INTO attendance
                    (student_id, attendance_date, attendance_time, status, confidence, created_at)
                VALUES (?, ?, ?, 'Present', ?, ?)
                """,
                (student_id, attendance_date, attendance_time, confidence, now.isoformat(timespec="seconds")),
            )
            connection.execute(
                """
                INSERT INTO recognition_logs
                    (student_id, recognized_name, log_date, log_time, result, confidence, status)
                VALUES (?, ?, ?, ?, 'Recognized', ?, 'Attendance marked')
                """,
                (student_id, student["name"], attendance_date, attendance_time, confidence),
            )
        return True, f"Present marked: {student['name']} ({student['roll_number']})."
    except sqlite3.IntegrityError:
        return False, "Attendance already recorded for today."
    except sqlite3.Error:
        return False, "Database error: attendance could not be recorded."


def get_attendance_records(limit: int | None = 200) -> list[dict]:
    """Return attendance records joined with student details, newest first."""
    query = """
        SELECT a.attendance_id, s.student_id, s.roll_number, s.name, s.class_name,
               s.division, s.department, a.attendance_date, a.attendance_time,
               a.status, a.confidence
        FROM attendance AS a
        JOIN students AS s ON s.student_id = a.student_id
        ORDER BY a.attendance_date DESC, a.attendance_time DESC
    """
    parameters = () if limit is None else (limit,)
    if limit is not None:
        query += " LIMIT ?"
    with get_connection() as connection:
        rows = connection.execute(query, parameters).fetchall()
    return [dict(row) for row in rows]
